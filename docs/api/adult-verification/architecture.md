## 1. Executive Summary

### 1.1 목적

TeddyBear's Room (성인용품 E-commerce)에 **KG이니시스 통합본인인증**을 연동하여 법적 요건을 충족하는 성인인증 시스템을 구축합니다.

### 1.2 핵심 요구사항

| 요구사항 | 설명 |
|---------|------|
| **법적 준수** | 청소년보호법에 따른 성인 확인 의무 |
| **중복 방지** | CI(연계정보) 해시로 1인 1계정 정책 |
| **OAuth 통합** | 기존 Supabase Auth (Kakao/Google) 플로우 유지 |
| **UX 최적화** | 최소한의 인증 단계, 팝업 방식 선호 |

### 1.3 선택된 솔루션

- **인증 제공자**: KG이니시스 통합본인인증 (PASS 포함)
- **인증 시점**: 회원가입 완료 직후 (OAuth 성공 후)
- **인증 방식**: 팝업 + Fallback Redirect
- **비용**: 건당 40원 (인증 성공 시에만 과금)

### 6.3 마이그레이션 스크립트

```sql
-- web/prisma/migrations/manual/add_adult_verification.sql

-- 1. Profile 테이블 확장
ALTER TABLE profiles
ADD COLUMN IF NOT EXISTS adult_verified BOOLEAN DEFAULT false,
ADD COLUMN IF NOT EXISTS adult_verified_at TIMESTAMPTZ,
ADD COLUMN IF NOT EXISTS ci_hash VARCHAR(64) UNIQUE,
ADD COLUMN IF NOT EXISTS birth_year INTEGER,
ADD COLUMN IF NOT EXISTS last_verify_error TEXT;

-- 2. 인덱스 추가
CREATE INDEX IF NOT EXISTS idx_profiles_ci_hash ON profiles(ci_hash);
CREATE INDEX IF NOT EXISTS idx_profiles_adult_verified ON profiles(adult_verified);

-- 3. 인증 로그 테이블 생성
CREATE TABLE IF NOT EXISTS verification_logs (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  profile_id UUID NOT NULL REFERENCES profiles(id) ON DELETE CASCADE,
  auth_type VARCHAR(20) NOT NULL,
  status VARCHAR(20) NOT NULL,
  error_code VARCHAR(50),
  error_msg TEXT,
  ip_address VARCHAR(45) NOT NULL,
  user_agent TEXT NOT NULL,
  tx_id VARCHAR(100),
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_verification_logs_profile
  ON verification_logs(profile_id, created_at DESC);
CREATE INDEX IF NOT EXISTS idx_verification_logs_status
  ON verification_logs(status);

-- 4. RLS 정책 (선택사항)
ALTER TABLE verification_logs ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Users can view own verification logs" ON verification_logs
  FOR SELECT USING (profile_id = auth.uid());

CREATE POLICY "Server can insert verification logs" ON verification_logs
  FOR INSERT WITH CHECK (true);

### 8.2 보안 구현 예시

```typescript
// lib/auth/inicis.ts - 보안 관련 함수

import crypto from 'crypto';

const INICIS_HMAC_KEY = process.env.INICIS_HMAC_KEY!;
const CI_HASH_SALT = process.env.CI_HASH_SALT!;

/**
 * 콜백 서명 검증
 */
export function verifyCallbackSignature(
  data: Record<string, string>,
  signature: string
): boolean {
  const sortedKeys = Object.keys(data).sort();
  const payload = sortedKeys
    .filter(k => k !== 'signature')
    .map(k => `${k}=${data[k]}`)
    .join('&');

  const expectedSig = crypto
    .createHmac('sha256', INICIS_HMAC_KEY)
    .update(payload)
    .digest('hex');

  return crypto.timingSafeEqual(
    Buffer.from(signature),
    Buffer.from(expectedSig)
  );
}

/**
 * CI 해시 생성 (솔트 포함)
 */
export function hashCI(ci: string): string {
  return crypto
    .createHash('sha256')
    .update(ci + CI_HASH_SALT)
    .digest('hex');
}

/**
 * Nonce 생성 (요청 고유값)
 */
export function generateNonce(): string {
  return crypto.randomBytes(16).toString('hex');
}

/**
 * 성인 여부 확인
 */
export function isAdult(birthDate: string): boolean {
  // birthDate: "19900515" 형식
  const birthYear = parseInt(birthDate.substring(0, 4));
  const birthMonth = parseInt(birthDate.substring(4, 6));
  const birthDay = parseInt(birthDate.substring(6, 8));

  const today = new Date();
  const age = today.getFullYear() - birthYear;

  // 생일 지났는지 확인
  const hasBirthdayPassed =
    today.getMonth() + 1 > birthMonth ||
    (today.getMonth() + 1 === birthMonth && today.getDate() >= birthDay);

  const actualAge = hasBirthdayPassed ? age : age - 1;

  return actualAge >= 19; // 한국 성인 기준 (만 19세)
}
```
